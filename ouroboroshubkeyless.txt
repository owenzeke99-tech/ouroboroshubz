-- Cấu hình Key, Link & File lưu trữ cache
local CORRECT_KEY = "OuroborosHubKeyBsksi - Kalapwowjenodoeen"
local GET_KEY_LINK = "https://link4m.org/pox55dP"
local CAFE_FILE_NAME = "Ouroboros_KeyCache.json"
local CACHE_DURATION = 24 * 60 * 60 -- 24 giờ tính bằng giây

-- Hàm kiểm tra và đọc cache đã lưu (nếu executor hỗ trợ writefile/readfile)
local function checkSavedKey()
	if writefile and readfile and isfile and isfile(CAFE_FILE_NAME) then
		local success, data = pcall(function()
			return game:GetService("HttpService"):JSONDecode(readfile(CAFE_FILE_NAME))
		end)
		if success and type(data) == "table" and data.key == CORRECT_KEY then
			if os.time() - data.timestamp < CACHE_DURATION then
				return true -- Key còn hiệu lực trong vòng 24h
			end
		end
	end
	return false
end

-- Hàm lưu key vào file cache
local function saveKeyCache()
	if writefile then
		pcall(function()
			local data = {
				key = CORRECT_KEY,
				timestamp = os.time()
			}
			writefile(CAFE_FILE_NAME, game:GetService("HttpService"):JSONEncode(data))
		end)
	end
end

-- Hàm thực thi load script chính
local function loadMainScript()
	local success, err = pcall(function()
		loadstring(game:HttpGet("https://raw.githubusercontent.com/joustingmatch/Ouroboros/main/loader.lua"))()
	end)
	if not success then
		warn("Error loading main script: " .. tostring(err))
	end
end

-- Nếu đã lưu key hợp lệ trong 24h thì tự động bỏ qua menu, chạy thẳng script
if checkSavedKey() then
	loadMainScript()
	return
end

-- Hiệu ứng Lắc mượt mà (Smooth Damped Sin Shake)
local function shakeFrame()
	local originalPos = UDim2.new(0.5, -210, 0.5, -125)
	local duration = 0.35
	local elapsed = 0
	local intensity = 12
	
	local connection
	connection = RunService.RenderStepped:Connect(function(dt)
		elapsed = elapsed + dt
		if elapsed >= duration then
			MainFrame.Position = originalPos
			connection:Disconnect()
		else
			local damping = 1 - (elapsed / duration)
			local offsetX = math.sin(elapsed * 45) * intensity * damping
			MainFrame.Position = originalPos + UDim2.new(0, offsetX, 0, 0)
		end
	end)
end
